## 批次刷新腳本（PowerShell）

```powershell
# 每季執行一次，驗證各資料集 API 仍可用
$datasets = @{
    "SITI通過案例" = "014c6c34-efdb-4aab-9715-50711f14562e"
    "老化指數"     = "aafb15dc-5508-4091-bd48-a708e60f6698"
    "商業登記"     = "5fdefcca-e0a6-41bc-a520-7c8f067caad3"
    "就業統計"     = "18f45255-8b16-4e6d-9a77-e48af4e497cd"
    "交通事故"     = "446eca64-c4f9-4a31-bf0b-db2765c7bb01"
    "新移民人口"   = "dc3e7c97-46bb-4e3f-b57c-0cd493048d3e"
}

$base = "https://data.taipei/api/v1/dataset"
foreach ($name in $datasets.Keys) {
    $id = $datasets[$name]
    try {
        $r = Invoke-RestMethod "$base/$id`?scope=resourceAquire&limit=1" -TimeoutSec 10
        Write-Host "✅ $name"
    } catch {
        Write-Host "❌ $name — $_"
    }
}
```
